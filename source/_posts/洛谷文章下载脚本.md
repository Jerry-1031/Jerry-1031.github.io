```javascript
(() => {
  const el = document.querySelector('#lentille-context');
  const article = el && JSON.parse(el.textContent)?.data?.article;
  const blob = new Blob([article.content], {
    type: 'text/markdown;charset=utf-8'
  });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = `${article.title}`.replace(/[<>:"/\\|?*\x00-\x1f]/g, '_') + '.md';
  document.body.appendChild(a);
  a.click();
  a.remove();
  setTimeout(() => URL.revokeObjectURL(url), 10000);
})();
```

浏览器打开洛谷博客文章页面，`F12` 打开控制台，粘贴进去即可下载。
